# Booklight 1.1 implementation plan

**Goal:** build everything in `docs/superpowers/specs/2026-10-01-booklight-1.1-design.md` as one release.

**Architecture:** `core/` (pure Kotlin, JUnit) gains scopes, verbs and the parsers; `app/` gains the
action row, scope chip, row bodies, the unfold opening, the Booklight window and the new effects.

**Spec:** `docs/superpowers/specs/2026-10-01-booklight-1.1-design.md`. Read with it:
`docs/design/design-system.md` (tokens, components, the motion table), `docs/design/ux-model.md` (keys),
`docs/research/device-findings.md` (what the devices allow), `docs/design/booklight-next.html` (how it
should feel).

**How it is executed:** in the main session, task by task, each committed working on `main`; two side
jobs run beside it (task 2 and task 3: pure Kotlin and a data file, nothing shared with the app code);
design and code reviewers at the end, as for 1.0.

## Global constraints

- Permissions are exactly: `INTERNET`, `REQUEST_DELETE_PACKAGES`, `com.android.alarm.permission.SET_ALARM`,
  `WRITE_SETTINGS`. Nothing else, no services that run in the background.
- Every user-facing string is a resource, in English and German. Copy is plain.
- The window is exactly the panel; `Metrics.height(model)` must equal what `Panel` draws.
- All motion specs come from `Motion.kt`; with the system's animations off everything cuts.
- minSdk 34, targetSdk 37. material3 stays pinned. One new dependency: `com.google.zxing:core`.
- No attribution lines in commits or code.
- Device rules in `CLAUDE.md` hold on the Lenovo as on the HP.

## Review focus

1. Text typed faster than the list: Enter must run a row of the *current* text and scope, never a stale one.
2. A scope keyword that is also an ordinary word (`new york weather`): the web row is there, Backspace
   restores the text, nothing is lost.
3. Destructive and outward actions: never armed by default, never run by an arrow key or Ctrl + digit;
   Delete needs the second Enter; nothing is sent anywhere without a window the user sees.
4. Missing handlers: no Gmail, no Clock, no Gemini, no calendar app, clipboard empty, notes folder revoked:
   the panel stays open and says so; no crash.
5. Animations off, dark theme, German, a narrow screen: nothing overlaps, nothing is clipped.

```kotlin
// Model.kt
data class Query(val raw: String, val scope: String? = null) { val text = raw.trim(); val isEmpty get() = text.isEmpty() }
enum class Kind { APP, ANSWER, SETTING, COMMAND, WEB, SUGGESTION, SCOPE, CONTROL, OTHER }
sealed interface Icon { App(packageName, className, user = 0); Symbol(name); Swatch(argb: Int); Glyph(text: String) }
enum class Place { NONE, LEFT, RIGHT, FULL }
enum class MediaKey { PLAY_PAUSE, NEXT, PREVIOUS }
enum class ImageUse { COPY, SAVE, SHARE }
sealed interface Effect {
    LaunchApp(packageName, className, user = 0, place = Place.NONE, newWindow = false)
    AppInfo(packageName, className, user = 0); StorePage(packageName); Uninstall(packageName, user = 0)
    OpenUrl(url); CopyText(text, sensitive = false); ShareText(text)
    OpenSettings(action); Internal(command)
    Compose(to: List<String>, subject: String, body: String)
    AppendNote(text); KeepNote(text)
    InsertEvent(title, startMillis: Long, endMillis: Long, allDay: Boolean, place: String)
    SetTimer(seconds: Int, label: String); SetAlarm(hour: Int, minute: Int, label: String)
    NewFile(name: String, folder: Boolean, pick: Boolean = false)
    AskGemini(text)
    SetVolume(percent: Int); ToggleMute; Media(key: MediaKey); PlayMusic(query)
    SetBrightness(percent: Int)
    QrImage(text, use: ImageUse)
    EnterScope(key: String, text: String = "")
    Grant(what: String)                    // "brightness", "notes"
    SaveSnippet(key: String, text: String); Delete(kind: String, id: String); Edit(kind: String, id: String)
    Steps(steps: List<Effect>)
}
data class Action(val id: String, val label: String, val effect: Effect, val keepOpen: Boolean = false,
    val symbol: String = id, val danger: Boolean = false, val confirm: Boolean = false, val done: String? = null)
sealed interface Body {
    data class Slots(val caption: String?, val slots: List<Slot>, val note: String? = null) : Body
    data class Level(val percent: Int, val target: Int? = null, val muted: Boolean = false, val locked: Boolean = false) : Body
    data object Media : Body
    data class Grid(val cells: List<Cell>, val columns: Int = 14) : Body
    data class Code(val text: String) : Body
    data class Mono(val text: String) : Body
}
enum class SlotState { TYPED, GUESSED, EMPTY }

data class Nudge(val down: Effect, val up: Effect)

    val armed: Int = 0, val body: Body? = null, val nudge: Nudge? = null)

// Scope.kt
interface Scope { val key: String; val keywords: List<String>; val name: String; val symbol: String; val hint: String
    suspend fun rows(arg: String): List<Result> }

// SearchEngine
class SearchEngine(providers, history, clock, budgetMs, private val scopes: () -> List<Scope> = { emptyList() },
    private val fallback: (String) -> List<Result> = { emptyList() })
    suspend fun search(q: Query, limit): List<Result>     // q.scope != null: that scope's rows, then fallback("keyword arg")
    fun scopeFor(text: String): Pair<Scope, String>?      // "yt lofi" → (yt, "lofi"); "yt " → (yt, ""); "yt" → null
    fun scope(key: String): Scope?
// outside a scope: rows that enter scopes come from the engine itself: Kind.SCOPE, id "scope:<key>",
// one action Effect.EnterScope(key); score = 1.0 for an exact keyword, 0.9 for a keyword prefix of ≥ 2 letters,
// Matcher.score(text, name) × 0.85 otherwise. Up to two WEB rows with score < URL_SCORE keep the end.
```

```kotlin
// Verbs.kt
data class Verb(val action: String, val words: List<String>, val min: Int = 2, val atStart: Boolean = true)
data class Reading(val rest: String, val action: String)
object Verbs { fun readings(text: String, verbs: List<Verb>): List<Reading> }
//  end: the last one or two words are a prefix (≥ verb.min letters, folded) of one of verb.words → rest = what precedes
//  start (only if verb.atStart): text starts with a whole verb word and a space → rest = what follows
//  "info" alone → no reading (a verb needs an object). Matching uses Matcher.fold.

// When.kt  (java.time; English and German)
data class Moment(val start: LocalDateTime, val end: LocalDateTime?, val allDay: Boolean,
    val dayGiven: Boolean, val timeGiven: Boolean, val rest: String)
object When { fun parse(text: String, now: LocalDateTime): Moment? }
//  finds a date and/or time expression at the START or the END of the text; rest = the other part, trimmed.
//  days: today tomorrow heute morgen übermorgen; weekday names and 2/3-letter forms (mon, mo, montag…): the next
//    such day, today if a time is given and still ahead; 3.10. 3.10.2026 10/3 2026-10-03 "oct 3" "3 oct" "3. okt"
//  times: 3pm 3:30pm 15:00 "15 uhr" 9-9:30 9am-10am 9–10 (an end time gives `end`); noon, midnight, mittag
//  relative: "in 20m" "in 2h" "in 3 days" "in 2 wochen" (timeGiven for m/h)
//  only a time: today if still ahead of now, else tomorrow (dayGiven = false). only a day: allDay = true.
//  nothing found → null.

// Jot.kt
data class MailDraft(val to: List<String>, val subject: String, val body: String)
data class EventDraft(val title: String, val start: LocalDateTime, val end: LocalDateTime, val allDay: Boolean,
    val place: String, val dayGiven: Boolean, val timeGiven: Boolean)
data class TimerSpec(val seconds: Int, val label: String)
data class AlarmSpec(val hour: Int, val minute: Int, val label: String)
sealed interface ReminderPlan { data class Timer(val spec: TimerSpec) : ReminderPlan
    data class Alarm(val spec: AlarmSpec, val at: LocalDateTime) : ReminderPlan
    data class Event(val draft: EventDraft) : ReminderPlan }
enum class NewKind { FILE, FOLDER, DOC, SHEET, SLIDES }
data class NewSpec(val kind: NewKind, val name: String, val kindGiven: Boolean)
object Jot {
    fun mail(arg: String): MailDraft          // leading words containing @ are recipients (commas stripped); " / " starts the body
    fun note(arg: String): String             // " / " becomes a line break
    fun event(arg: String, now: LocalDateTime): EventDraft   // "@ place" at the end; no When → today, all day, guessed; 1 h default
    fun reminder(arg: String, now: LocalDateTime): ReminderPlan?  // "in 20m x" → Timer; a time within 24 h → Alarm; later → Event; no time → null
    fun timer(arg: String): TimerSpec?        // "10" = minutes; 10m 90s 1h30 "1h 30m" "10 min" "1 std"; rest = label
    fun alarm(arg: String, now: LocalDateTime): AlarmSpec?   // "7:30 gym", "7am"
    fun new(arg: String): NewSpec             // "file x.md" "folder A/B" "doc Title" "ordner X" "datei x"; an extension means FILE
}

// Colors.kt
data class Rgb(val r: Int, val g: Int, val b: Int) { val argb: Int; fun hex(): String  // "#3478F6"
    fun rgb(): String /* "rgb(52, 120, 246)" */; fun hsl(): String /* "hsl(219, 91%, 58%)" */; fun oklch(): String /* "oklch(0.59 0.19 262)" */ }
object Colors { fun parse(text: String): Rgb? }   // #rgb #rrggbb rgb(…) hsl(…)

// Templates.kt
object Templates { fun takesArgument(link: String): Boolean
    fun fill(link: String, argument: String, clipboard: String, date: LocalDate): String }  // %s {argument} {query} {clipboard} {date}; values URL-encoded

// Clip.kt
data class Transform(val id: String, val value: String)   // ids: upper lower title line count url
object Clip { fun transforms(text: String): List<Transform> }

// Secrets.kt
object Secrets { fun password(length: Int, random: java.util.Random): String   // 8..64; letters without lookalikes, digits, symbols; at least one of each
    fun uuid(random: java.util.Random): String }

// Emoji.kt   (table: glyph \t kind e|s \t name en \t words en \t name de \t words de; words separated by "|")
data class Emoji(val glyph: String, val symbol: Boolean, val name: String, val words: List<String>)
class EmojiIndex(lines: Sequence<String>, german: Boolean) {
    fun search(text: String, symbols: Boolean, limit: Int): List<Emoji>   // name prefix > word prefix > inside; table order breaks ties; English names also match in German
    fun first(symbols: Boolean, limit: Int): List<Emoji>
    fun find(glyph: String): Emoji? }

// Jumps.kt
object Jumps { fun github(arg: String): String?   // "owner/repo" → https://github.com/owner/repo ; "owner/repo#12" → …/issues/12
    fun port(text: String): String? }             // ":3000" → http://localhost:3000 ; ":3000/x" keeps the path

// Ask.kt
object Ask { fun first(text: String): Boolean     // ≥ 5 words, or ends with "?", or a question word (EN/DE) first and ≥ 3 words
    fun offered(text: String): Boolean }          // ≥ 2 words
```
